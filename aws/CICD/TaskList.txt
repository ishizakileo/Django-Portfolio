[3]起動テンプレートの最新化(buildspec.ymlとScript)
  # buildspec.yml
  version: 0.2
  phases:
    post_build:
      commands:
        # デプロイ成功確認
        - |
          if [ "${CODEBUILD_BUILD_SUCCEEDING}" = "1" ]; then
            # 現在のインスタンスからAMIを作成
            AMI_ID=$(aws ec2 create-image \
              --instance-id ${INSTANCE_ID} \
              --name "app-${ENV}-$(date +%Y%m%d-%H%M%S)" \
              --description "Application AMI for ${ENV} environment" \
              --no-reboot \
              --query 'ImageId' \
              --output text)
            
            # AMI作成完了を待機
            aws ec2 wait image-available --image-ids ${AMI_ID}
            
            # 起動テンプレートの新バージョンを作成
            aws ec2 create-launch-template-version \
              --launch-template-id ${LAUNCH_TEMPLATE_ID} \
              --source-version 1 \
              --launch-template-data "{\"ImageId\":\"${AMI_ID}\"}"
              
            # 新バージョンをデフォルトに設定
            aws ec2 modify-launch-template \
              --launch-template-id ${LAUNCH_TEMPLATE_ID} \
              --default-version '$Latest'
              
            # 古いAMIのクリーンアップ（オプション）
            ./cleanup_old_amis.sh
          fi

  # cleanup_old_amis.sh
  #!/bin/bash

  # 保持するAMIの数
  KEEP_IMAGES=5

  # アプリケーションAMIの一覧を取得
  IMAGES=$(aws ec2 describe-images \
    --owners self \
    --filters "Name=name,Values=app-${ENV}-*" \
    --query 'Images[*].[ImageId,CreationDate,Name]' \
    --output text | sort -k2 -r)

  # 古いAMIを削除
  COUNT=0
  echo "$IMAGES" | while read IMAGE_ID CREATION_DATE NAME; do
    COUNT=$((COUNT + 1))
    if [ $COUNT -gt $KEEP_IMAGES ]; then
      echo "Deregistering AMI: $IMAGE_ID ($NAME)"
      aws ec2 deregister-image --image-id $IMAGE_ID
    fi
  done

[4]ロールバック手法
  ### 推奨するロールバックフロー
  1. **即時対応が必要な場合**
  - 直前のバージョンへの即時ロールバック
  - スクリプトによる自動化
  - 最小限の承認プロセス

  2. **計画的な対応が可能な場合**
  - GitLabを経由した通常のデプロイフロー
  - コード修正とレビュー
  - テスト実行による品質確保

  1. **ECSの場合**
  ```bash
  # 即時ロールバック用スクリプト
  #!/bin/bash

  # 現在のサービス情報を取得
  CURRENT_SERVICE=$(aws ecs describe-services \
    --cluster ${CLUSTER_NAME} \
    --services ${SERVICE_NAME})

  # 1つ前のタスク定義を取得
  PREVIOUS_TASK_DEFINITION=$(echo $CURRENT_SERVICE | \
    jq -r '.services[0].taskDefinition' | \
    sed -e 's/:[0-9]\+/:'"$(($(echo $CURRENT_SERVICE | \
    jq -r '.services[0].taskDefinition' | \
    cut -d: -f2) - 1))"'/')

  # ロールバックを実行
  aws ecs update-service \
    --cluster ${CLUSTER_NAME} \
    --service ${SERVICE_NAME} \
    --task-definition $PREVIOUS_TASK_DEFINITION \
    --force-new-deployment

  2. **EC2の場合**
  ```bash
  # 即時ロールバック用スクリプト
  #!/bin/bash

  # 直前のデプロイメントIDを取得
  PREVIOUS_DEPLOYMENT=$(aws deploy list-deployments \
    --application-name ${APPLICATION_NAME} \
    --deployment-group-name ${DEPLOYMENT_GROUP} \
    --include-only-statuses Succeeded \
    --max-items 2 | \
    jq -r '.deployments[1]')

  # ロールバックデプロイを作成
  aws deploy create-deployment \
    --application-name ${APPLICATION_NAME} \
    --deployment-group-name ${DEPLOYMENT_GROUP} \
    --revision revisionType=S3,s3Location="{bucket=${BUCKET},key=${KEY},bundleType=zip,version=${VERSION}}"
